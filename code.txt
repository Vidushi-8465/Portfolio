  <script>
    // ===== Utilities =====
    const $ = s => document.querySelector(s);
    const $$ = s => Array.from(document.querySelectorAll(s));

    // Update year
    $('#year').textContent = new Date().getFullYear();

    // Active nav link on scroll
    const sections = ['home','about','projects','education','contact'].map(id=>document.getElementById(id));
    const navLinks = $$('.nav-link');
    const ioNav = new IntersectionObserver((entries)=>{
      entries.forEach(e=>{
        if(e.isIntersecting){
          navLinks.forEach(a=>a.classList.remove('active'));
          const active = document.querySelector(`a[href="#${e.target.id}"]`);
          active && active.classList.add('active');
        }
      })
    },{rootMargin:'-50% 0px -50% 0px',threshold:0});
    sections.forEach(sec=>ioNav.observe(sec));

    // Simple scroll reveal
    const io = new IntersectionObserver((entries)=>{
      entries.forEach(e=>{ if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target);} });
    },{threshold:.2});
    $$('.reveal, .project-card, .t-item').forEach(el=>io.observe(el));

    // Timeline animation (individual cards)
    const ioTime = new IntersectionObserver((entries)=>{
      entries.forEach(e=>{ if(e.isIntersecting){ e.target.classList.add('reveal'); ioTime.unobserve(e.target); }});
    },{threshold:.35});
    $$('.t-item').forEach(el=>ioTime.observe(el));

    // ===== Project Modals =====
    const modal = document.createElement('div');
    modal.className='modal';
    modal.innerHTML=`<div class="modal-card" role="dialog" aria-modal="true">
        <div class="modal-header">
          <strong id="modal-title">Project</strong>
          <button class="close" id="modal-close">Close</button>
        </div>
        <div class="modal-body" id="modal-body"></div>
      </div>`;
    document.body.appendChild(modal);
    $('#modal-close').addEventListener('click',()=>modal.classList.remove('open'));
    modal.addEventListener('click',e=>{ if(e.target===modal) modal.classList.remove('open'); });

    const projectDetails = {
      p1: {
        title: 'TaskFlow — Kanban task manager',
        html: `<p>Full‑stack app with column swimlanes, drag‑and‑drop via <b>SortableJS</b>, and optimistic UI updates. Real‑time board sync using <b>Socket.io</b>. Deployed on Vercel with a Prisma + MongoDB backend.</p>
               <ul><li>Email/password & OAuth login</li><li>Role‑based permissions per board</li><li>Server actions for latency‑free UX</li></ul>`
      },
      p2: {
        title: 'Weatherly — Minimal weather dashboard',
        html: `<p>SPA that consumes the OpenWeather API and visualizes data with <b>Chart.js</b>. Features favourite cities, hourly forecasts, and a clean card UI with skeleton loaders.</p>`
      },
      p3: {
        title: 'AuthKit — Authentication boilerplate',
        html: `<p>Node/Express starter with <b>JWT</b> and cookie sessions, CSRF protection, rate limiting, password reset, and ready‑to‑use email templates.</p>`
      },
      p4: {
        title: 'DocuSpark — Documentation theme',
        html: `<p>Fast static docs with full‑text search (Algolia), MDX components, code copy buttons, and accessible dark/light themes.</p>`
      }
    };

    $$('.project-card').forEach(card=>{
      card.addEventListener('click',()=>{
        const id = card.dataset.project;
        const data = projectDetails[id];
        if(!data) return;
        $('#modal-title').textContent = data.title;
        $('#modal-body').innerHTML = data.html;
        modal.classList.add('open');
      });
    });

    // ===== Contact Form (EmailJS) =====
    /*  ⚠️ How to make email sending work:
        1) Create an account at https://www.emailjs.com/
        2) Add an Email Service and a Template with fields: name, email, phone, message
        3) Put your Public Key here (emailjs.init('PUBLIC_KEY'))
        4) Replace SERVICE_ID and TEMPLATE_ID below.
       This is 100% frontend and keeps this a "full working website frontend". */

    // Replace with your EmailJS Public Key
    emailjs.init('PUBLIC_KEY_HERE');

    $('#contact-form').addEventListener('submit', async (e)=>{
      e.preventDefault();
      const fd = new FormData(e.currentTarget);
      const payload = Object.fromEntries(fd.entries());
      const status = $('#form-status');
      status.textContent = 'Sending…';
      try{
        await emailjs.send('SERVICE_ID','TEMPLATE_ID', payload);
        status.textContent = 'Thanks! Your message has been sent.';
        e.currentTarget.reset();
      }catch(err){
        console.error(err);
        status.textContent = 'Could not send right now. Please check your EmailJS keys or try again later.';
      }
    });
  </script>
